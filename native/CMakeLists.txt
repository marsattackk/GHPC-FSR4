cmake_minimum_required(VERSION 3.20)
project(FSR4Bridge CXX)

add_library(FSR4Bridge SHARED bridge.cpp)
target_compile_features(FSR4Bridge PRIVATE cxx_std_17)
target_include_directories(FSR4Bridge PRIVATE ffx/Kits/FidelityFX/api/include ffx/Kits/FidelityFX/upscalers/include)
target_compile_definitions(FSR4Bridge PRIVATE _WINDOWS WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(FSR4Bridge PRIVATE /W3 /EHa /MP)
target_link_libraries(FSR4Bridge PRIVATE d3d11 d3d12 dxgi)
set_property(TARGET FSR4Bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
option(FSR4_PUBLIC "Public build: dev-only exports and code left out" OFF)
if(FSR4_PUBLIC)
  target_compile_definitions(FSR4Bridge PRIVATE PUBLIC=1)
  target_link_options(FSR4Bridge PRIVATE /DEBUG:NONE /NOCOFFGRPINFO /NOVCFEATURE)
endif()
